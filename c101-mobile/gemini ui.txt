    // --- TAB 5: AI TENANT ASSISTANT CHAT ---
    function ChatTab({ rentData }) {
      const [messages, setMessages] = useState([
        { id: 1, sender: 'bot', text: 'Merhaba! Ben Kiracı Hakları Asistanınız. Türk Borçlar Kanunu çerçevesinde kira zamları, tahliye taahhüdü, kombi/tesisat masrafları ve depozito hakkında bana soru sorabilirsiniz.' }
      ]);
      const [input, setInput] = useState('');
      const chatEndRef = useRef(null);

      useEffect(() => {
        chatEndRef.current?.scrollIntoView({ behavior: 'smooth' });
      }, [messages]);

      const handleSend = (e) => {
        e.preventDefault();
        if (!input.trim()) return;

        const userMsg = input;
        setMessages(prev => [...prev, { id: Date.now(), sender: 'user', text: userMsg }]);
        setInput('');

        setTimeout(() => {
          let botReply = 'Kira mevzuatı çerçevesinde konuyu inceledim. TBK 344 uyarınca ev sahibinin talepleri yasal sınırları aşamaz. Haklarınızı korumak için yazılı ihtar çekebilirsiniz.';
          
          const lower = userMsg.toLowerCase();
          if (lower.includes('zam') || lower.includes('artış') || lower.includes('tüfe')) {
            botReply = `Yasal zam ceiling %${rentData.tufeRate} (12 Aylık TÜFE ortalaması) ile sınırlıdır. Mevcut kiranız olan ₺${rentData.amount.toLocaleString('tr-TR')} için en fazla ₺${Math.round(rentData.amount * (1 + rentData.tufeRate/100)).toLocaleString('tr-TR')} talep edilebilir.`;
          } else if (lower.includes('kombi') || lower.includes('tamir') || lower.includes('masraf')) {
            botReply = 'Kombi arızası, boru patlaması ve çatı sızıntısı gibi kullanım hatasından kaynaklanmayan ana demirbaş masrafları tamamen EV SAHİBİNE aittir (TBK Madde 301). Faturayı kiradan düşebilirsiniz.';
          } else if (lower.includes('çık') || lower.includes('tahliye') || lower.includes('oğlum')) {
            botReply = 'Ev sahibi keyfi olarak sizi çıkaramaz. İhtiyaç sebebiyle tahliyelerde mahkeme somut delil arar ve ev boşaldığında 3 yıl başkasına kiralama yasağı gelir (TBK Madde 355).';
          } else if (lower.includes('depozito')) {
            botReply = 'Depozito en fazla 3 aylık kira kadar olabilir ve ortak vadeli hesapta tutulmalıdır. Hasarsız teslimde faiziyle iade edilmek zorundadır (TBK Madde 342).';
          }

          setMessages(prev => [...prev, { id: Date.now() + 1, sender: 'bot', text: botReply }]);
        }, 700);
      };

      return (
        <div className="h-[520px] flex flex-col justify-between animate-fadeIn bg-white rounded-2xl shadow-card-soft border border-slate-100 overflow-hidden">
          
          {/* Chat Header */}
          <div className="bg-slate-900 text-white p-3 text-xs flex items-center justify-between shrink-0 border-b border-slate-800">
            <div className="flex items-center gap-2.5">
              <div className="w-8 h-8 rounded-full bg-brand-600 flex items-center justify-center text-sm shadow">🤖</div>
              <div>
                <p className="font-extrabold text-white">Yapay Zeka Kiracı Danışmanı</p>
                <p className="text-[10px] text-emerald-400 font-semibold">Çevrimiçi • TBK Mevzuat Uyumlu</p>
              </div>
            </div>
          </div>

          {/* Messages Area */}
          <div className="flex-1 p-3 overflow-y-auto space-y-2.5 bg-warmBg">
            {messages.map((m) => (
              <div key={m.id} className={`flex ${m.sender === 'user' ? 'justify-end' : 'justify-start'}`}>
                <div className={`max-w-[84%] p-3 rounded-2xl text-xs leading-relaxed ${m.sender === 'user' ? 'bg-brand-700 text-white rounded-br-none shadow' : 'bg-white text-slate-800 shadow-sm border border-slate-200/80 rounded-bl-none'}`}>
                  {m.text}
                </div>
              </div>
            ))}
            <div ref={chatEndRef} />
          </div>

          {/* Quick Questions Chips */}
          <div className="p-2 bg-white border-t border-slate-100 flex gap-1.5 overflow-x-auto no-scrollbar shrink-0">
            <button onClick={() => setInput('Kombi tamir masrafını kim öder?')} className="whitespace-nowrap bg-slate-100 hover:bg-slate-200 text-slate-700 text-[10px] font-bold px-2.5 py-1 rounded-full active:scale-95 transition">
              🔧 Kombi Masrafı
            </button>
            <button onClick={() => setInput('Ev sahibi beni evden çıkarabilir mi?')} className="whitespace-nowrap bg-slate-100 hover:bg-slate-200 text-slate-700 text-[10px] font-bold px-2.5 py-1 rounded-full active:scale-95 transition">
              🚪 Evden Çıkarma
            </button>
            <button onClick={() => setInput('Yasal zam oranım nedir?')} className="whitespace-nowrap bg-slate-100 hover:bg-slate-200 text-slate-700 text-[10px] font-bold px-2.5 py-1 rounded-full active:scale-95 transition">
              📈 Zam Ceiling
            </button>
          </div>

          {/* Input Box */}
          <form onSubmit={handleSend} className="p-2.5 bg-white border-t border-slate-200/80 flex gap-2 shrink-0">
            <input 
              type="text" 
              placeholder="Hukuki sorunuzu yazın..." 
              value={input}
              onChange={(e) => setInput(e.target.value)}
              className="flex-1 bg-warmBg border border-slate-200 rounded-xl px-3 py-2 text-xs font-medium text-slate-800 focus:outline-none focus:border-brand-600"
            />
            <button type="submit" className="w-9 h-9 rounded-xl bg-brand-700 hover:bg-brand-800 text-white flex items-center justify-center shrink-0 shadow active:scale-95 transition">
              <Icon name="send" className="w-4 h-4" />
            </button>
          </form>

        </div>
      );
    }
